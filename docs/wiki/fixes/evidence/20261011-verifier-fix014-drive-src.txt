"""FIX-014 verifier 드라이버 — bash-write-guard.sh 를 stdin JSON 으로 직접 호출해 deny/allow 표를 만든다."""
import json, os, subprocess, sys

ROOT = sys.argv[1]
SP = sys.argv[2]
HOOK = os.path.join(ROOT, ".claude", "hooks", "bash-write-guard.sh")
ENV = dict(os.environ, CLAUDE_PROJECT_DIR=ROOT)


def run(cmd, tool="Bash"):
    payload = json.dumps({"tool_name": tool, "tool_input": {"command": cmd}})
    p = subprocess.run(["bash", HOOK], input=payload.encode(), capture_output=True, env=ENV, timeout=30)
    out = p.stdout.decode("utf-8", "replace").strip()
    if not out:
        return "allow", ""
    try:
        d = json.loads(out)
        dec = d["hookSpecificOutput"]["permissionDecision"]
        reason = d["hookSpecificOutput"].get("permissionDecisionReason", "")
        return dec, reason
    except Exception:
        return "??", out[:200]

def table(title, rows):
    mism = 0
    print(f"## {title}")
    print("| # | 라벨 | 툴 | 기대 | 실제 | 판정 | 사유(앞 90자) |")
    print("|---|---|---|---|---|---|---|")
    for i, (label, tool, cmd, exp) in enumerate(rows, 1):
        dec, reason = run(cmd, tool)
        if exp == "limit":
            mark = "한계(막지 않음)" if dec == "allow" else "deny(한계인데도 막힘)"
        else:
            ok = (dec == exp)
            mark = "일치" if ok else "**불일치**"
            if not ok:
                mism += 1
        r = reason.replace("|", "\\|").replace("\n", " ")[:90]
        lab = label.replace("|", "\\|")
        print(f"| {i} | {lab} | {tool} | {exp} | {dec} | {mark} | {r} |")
    print()
    return mism
