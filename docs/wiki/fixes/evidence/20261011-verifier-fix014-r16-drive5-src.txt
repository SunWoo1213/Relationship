def run(cmd, tool="Bash", raw=None):
    payload = raw if raw is not None else json.dumps({"tool_name": tool, "tool_input": {"command": cmd}})
    p = subprocess.run(["/bin/bash", HOOK], input=payload.encode(), capture_output=True, env=ENV, timeout=30)
    out = p.stdout.decode("utf-8", "replace").strip()
    if not out:
        return "allow", f"(stdout 없음, rc={p.returncode})"
    try:
        d = json.loads(out)
        dec = d["hookSpecificOutput"]["permissionDecision"]
        return dec, d["hookSpecificOutput"].get("permissionDecisionReason", "")
    except Exception:
        return "??", out[:200]

def table(title, cases):
    print(f"\n## {title}")
    print("| # | 라벨 | 기대 | 실제 | 판정 | 사유(앞 110자) |")
    print("|---|---|---|---|---|---|")
    mism = []
    for i, (label, cmd, exp) in enumerate(cases, 1):
        dec, r = run(cmd)
        ok = "일치" if dec == exp else "**불일치**"
        if dec != exp:
            mism.append(label)
        print(f"| {i} | {label} | {exp} | {dec} | {ok} | {r[:110].replace('|', '¦')} |")
    print(f"{title}: 불일치 {len(mism)}" + (f" — {', '.join(mism)}" if mism else ""))
    return len(mism)
