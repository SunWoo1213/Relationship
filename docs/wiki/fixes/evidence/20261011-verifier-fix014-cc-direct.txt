## 스크립트 원문 (cc_test.sh — 재실행용)
#!/usr/bin/env bash
set -u
R=/Users/sunwoo/Desktop/Portfolio/Relationship
SP=<스크래치패드 절대경로>
T="$SP/cc-iso"
rm -f "$T"/.claude/.commit-approved "$T"/.claude/.push-approved "$T"/.claude/.awaiting-decision 2>/dev/null
mkdir -p "$T/.claude/hooks" "$T/.claude/scripts" "$T/docs/wiki"
cp "$R/.claude/hooks/commit-cleanup.sh" "$R/.claude/hooks/_py.sh" "$T/.claude/hooks/"
cp "$R/.claude/scripts/bash_write_check.py" "$T/.claude/scripts/"
printf '# journal\n' > "$T/docs/wiki/journal.md"
if [ ! -d "$T/.git" ]; then
  ( cd "$T" && git init -q . && git config user.email t@example.com && git config user.name tester \
    && printf 'x\n' > f.txt && git add f.txt && git commit -qm "격리 첫 커밋" ) >/dev/null 2>&1
fi
run() { printf '{"tool_name":"Bash","tool_input":{"command":%s}}' "$(python3 -c 'import json,sys;print(json.dumps(sys.argv[1]))' "$1")" \
        | CLAUDE_PROJECT_DIR="$T" bash "$T/.claude/hooks/commit-cleanup.sh" >/dev/null 2>&1; }
cnt() { c="$(grep -c "$1" "$T/docs/wiki/journal.md" 2>/dev/null)" || c="${c:-0}"; printf '%s' "${c:-0}"; }
state() { printf 'COMMIT=%s PUSH-dev2=%s PUSH=%s | commit-approved=%s push-approved=%s awaiting=%s\n' \
  "$(cnt '| COMMIT |')" "$(cnt 'PUSH-dev2')" "$(cnt '| PUSH |')" \
  "$([ -f "$T/.claude/.commit-approved" ] && echo yes || echo no)" \
  "$([ -f "$T/.claude/.push-approved" ] && echo yes || echo no)" \
  "$([ -f "$T/.claude/.awaiting-decision" ] && echo yes || echo no)"; }
sync_ref() { ( cd "$T" && git update-ref "refs/remotes/origin/$1" "$(git rev-parse HEAD)" ) 2>/dev/null; }
sync_ref dev2; sync_ref dev
( cd "$T" && git log -1 --format=%s ) > "$T/.claude/commit-draft.txt"
printf 'x\n' > "$T/.claude/.commit-approved"
printf 'x\n' > "$T/.claude/.push-approved"
echo "[준비] $(state)"
echo "--- 인자 속 글자 (기대: 전부 변화 없음) ---"
run 'python3 -c "print(\"git push origin dev2\")"';          echo "python -c 인자 'git push origin dev2'     → $(state)"
run 'grep "git commit" docs/wiki/journal.md';                echo "grep 인자 'git commit'                   → $(state)"
run 'echo "git push origin dev2:dev"';                       echo "echo 인자 'git push origin dev2:dev'     → $(state)"
run "printf '%s' 'git commit -F .claude/commit-draft.txt'";  echo "printf 인자 'git commit -F …'           → $(state)"
run 'sed -n "/git push origin dev2/p" x';                    echo "sed 스크립트 인자                       → $(state)"
echo "--- 실제 git 조각 (기대: 기존처럼 처리) ---"
run 'git push origin dev2';                                  echo "git push origin dev2                      → $(state)   [기대 PUSH-dev2 +1]"
run 'echo hi; git push origin dev2';                         echo "echo hi; git push origin dev2             → $(state)   [기대 PUSH-dev2 +1]"
run 'git commit -F .claude/commit-draft.txt';                echo "git commit -F …                           → $(state)   [기대 COMMIT +1, commit-approved no]"
run 'git push origin dev2:dev';                              echo "git push origin dev2:dev                  → $(state)   [기대 PUSH +1, push-approved no, awaiting yes]"
echo "--- journal ---"; cat "$T/docs/wiki/journal.md"
