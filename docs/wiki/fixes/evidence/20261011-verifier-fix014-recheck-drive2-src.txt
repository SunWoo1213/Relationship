bash-write-guard.sh 를 stdin JSON 으로 직접 호출한다. 기대 열: deny / allow / limit(수정안 5 한계 — 기록만)."""
import json, os, subprocess, sys

ROOT = sys.argv[1]

HOOK = os.path.join(ROOT, ".claude", "hooks", "bash-write-guard.sh")
ENV = dict(os.environ, CLAUDE_PROJECT_DIR=ROOT)


def run(cmd, tool="Bash", raw=None, env=None):
    payload = raw if raw is not None else json.dumps({"tool_name": tool, "tool_input": {"command": cmd}})
    p = subprocess.run(["/bin/bash", HOOK], input=payload.encode(), capture_output=True, env=env or ENV, timeout=30)
    out = p.stdout.decode("utf-8", "replace").strip()
    if not out:
        return "allow", f"(stdout 없음, rc={p.returncode})"
    try:
        d = json.loads(out)
        dec = d["hookSpecificOutput"]["permissionDecision"]
        reason = d["hookSpecificOutput"].get("permissionDecisionReason", "")
        return dec, reason
    except Exception:
        return "??", out[:200]

def table(title, rows, tool="Bash"):
    mism = 0
    print(f"## {title}")
    print("| # | 라벨 | 기대 | 실제 | 판정 | 사유(앞 110자) |")
    print("|---|---|---|---|---|---|")
    for label, cmd, exp in rows:
        dec, reason = run(cmd, tool)
        if exp == "limit":
            mark = "한계(막지 않음)" if dec == "allow" else "deny(막힘)"
        else:
            ok = (dec == exp)
            mark = "일치" if ok else "**불일치**"
            if not ok:
                mism += 1
        r = reason.replace("|", "\\|").replace("\n", " ")[:110]
        lab = label.replace("|", "\\|")
        print(f"| {lab} | {exp} | {dec} | {mark} | {r} |")
    print()
    return mism
