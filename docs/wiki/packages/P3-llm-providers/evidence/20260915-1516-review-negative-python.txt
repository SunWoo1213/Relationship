def case(name: str, ok: bool, detail: str) -> None:
    results.append((name, ok, detail))
    print(f"{'OK  ' if ok else 'MISS'} {name}: {detail}")


def expect_invalid(name: str, fn, must_contain=(), must_not_contain=()):
    try:
        fn()
    except InvalidValue as exc:
        msg = str(exc)
        ok = all(s in msg for s in must_contain) and not any(s in msg for s in must_not_contain)
        case(name, ok, f"InvalidValue: {msg}")
    except Exception as exc:  # noqa: BLE001
        case(name, False, f"wrong exception {type(exc).__name__}: {exc}")
    else:
        case(name, False, "no exception")
