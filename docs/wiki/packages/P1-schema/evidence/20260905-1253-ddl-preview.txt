===== persons =====

CREATE TABLE persons (
	id BIGINT GENERATED BY DEFAULT AS IDENTITY, 
	user_id TEXT NOT NULL, 
	display_name TEXT NOT NULL, 
	relation_tag TEXT NOT NULL, 
	hierarchy TEXT NOT NULL, 
	created_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL, 
	updated_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL, 
	CONSTRAINT pk_persons PRIMARY KEY (id), 
	CONSTRAINT ck_persons_relation_tag_valid CHECK (relation_tag IN ('가족', '연인', '친구', '직장', '지인')), 
	CONSTRAINT ck_persons_hierarchy_valid CHECK (hierarchy IN ('상', '동', '하'))
)



===== person_aliases =====

CREATE TABLE person_aliases (
	id BIGINT GENERATED BY DEFAULT AS IDENTITY, 
	person_id BIGINT NOT NULL, 
	alias TEXT NOT NULL, 
	source TEXT NOT NULL, 
	embedding VECTOR(1536), 
	confirmed_at TIMESTAMP WITH TIME ZONE, 
	CONSTRAINT pk_person_aliases PRIMARY KEY (id), 
	CONSTRAINT fk_person_aliases_person_id_persons FOREIGN KEY(person_id) REFERENCES persons (id) ON DELETE CASCADE
)



===== person_facts =====

CREATE TABLE person_facts (
	id BIGINT GENERATED BY DEFAULT AS IDENTITY, 
	person_id BIGINT NOT NULL, 
	key TEXT NOT NULL, 
	value TEXT NOT NULL, 
	confidence DOUBLE PRECISION NOT NULL, 
	updated_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL, 
	CONSTRAINT pk_person_facts PRIMARY KEY (id), 
	CONSTRAINT fk_person_facts_person_id_persons FOREIGN KEY(person_id) REFERENCES persons (id) ON DELETE CASCADE
)



===== events =====

CREATE TABLE events (
	id BIGINT GENERATED BY DEFAULT AS IDENTITY, 
	person_id BIGINT NOT NULL, 
	type TEXT NOT NULL, 
	content TEXT NOT NULL, 
	raw_utterance TEXT NOT NULL, 
	occurred_at TIMESTAMP WITH TIME ZONE NOT NULL, 
	created_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL, 
	CONSTRAINT pk_events PRIMARY KEY (id), 
	CONSTRAINT ck_events_type_valid CHECK (type IN ('conflict', 'praise', 'meal', 'meeting', 'personal_share', 'favor', 'other')), 
	CONSTRAINT fk_events_person_id_persons FOREIGN KEY(person_id) REFERENCES persons (id) ON DELETE CASCADE
)



===== fact_sources =====

CREATE TABLE fact_sources (
	fact_id BIGINT NOT NULL, 
	event_id BIGINT NOT NULL, 
	CONSTRAINT pk_fact_sources PRIMARY KEY (fact_id, event_id), 
	CONSTRAINT fk_fact_sources_fact_id_person_facts FOREIGN KEY(fact_id) REFERENCES person_facts (id) ON DELETE CASCADE, 
	CONSTRAINT fk_fact_sources_event_id_events FOREIGN KEY(event_id) REFERENCES events (id) ON DELETE CASCADE
)



===== schedules =====

CREATE TABLE schedules (
	id BIGINT GENERATED BY DEFAULT AS IDENTITY, 
	person_id BIGINT NOT NULL, 
	title TEXT NOT NULL, 
	scheduled_at TIMESTAMP WITH TIME ZONE NOT NULL, 
	briefed_at TIMESTAMP WITH TIME ZONE, 
	CONSTRAINT pk_schedules PRIMARY KEY (id), 
	CONSTRAINT fk_schedules_person_id_persons FOREIGN KEY(person_id) REFERENCES persons (id) ON DELETE CASCADE
)



===== pending_questions =====

CREATE TABLE pending_questions (
	id BIGINT GENERATED BY DEFAULT AS IDENTITY, 
	session_id TEXT NOT NULL, 
	kind TEXT NOT NULL, 
	question TEXT NOT NULL, 
	options JSONB NOT NULL, 
	context JSONB NOT NULL, 
	answer TEXT, 
	created_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL, 
	answered_at TIMESTAMP WITH TIME ZONE, 
	CONSTRAINT pk_pending_questions PRIMARY KEY (id), 
	CONSTRAINT ck_pending_questions_kind_valid CHECK (kind IN ('identity', 'new_person', 'schedule'))
)



===== push_subscriptions =====

CREATE TABLE push_subscriptions (
	id BIGINT GENERATED BY DEFAULT AS IDENTITY, 
	user_id TEXT NOT NULL, 
	endpoint TEXT NOT NULL, 
	keys JSONB NOT NULL, 
	created_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL, 
	CONSTRAINT pk_push_subscriptions PRIMARY KEY (id)
)



===== agent_traces =====

CREATE TABLE agent_traces (
	id BIGINT GENERATED BY DEFAULT AS IDENTITY, 
	session_id TEXT NOT NULL, 
	step TEXT NOT NULL, 
	tool_name TEXT NOT NULL, 
	input JSONB NOT NULL, 
	output JSONB NOT NULL, 
	tokens_in INTEGER NOT NULL, 
	tokens_out INTEGER NOT NULL, 
	created_at TIMESTAMP WITH TIME ZONE DEFAULT now() NOT NULL, 
	CONSTRAINT pk_agent_traces PRIMARY KEY (id)
)
