  let reject!: (e: unknown) => void
  const firstCall = new Promise<Response>((_, rej) => {
    reject = rej
  })
  let n = 0
  const fetchFn = vi.fn((input: RequestInfo | URL) => {
    const url = String(input)
    if (url === '/api/persons/3') return Promise.resolve(json(CARD))
    if (url === '/api/events/9/raw') {
      n++
      return n === 1 ? firstCall : Promise.resolve(json(RAW))
    }
    throw new Error('예상하지 못한 요청: ' + url)
  })
  vi.stubGlobal('fetch', fetchFn)
  render(<PersonCardScreen personId={3} />)
  await screen.findByTestId('person-card')
  const item = screen.getByTestId('timeline-9')
  const user = userEvent.setup()
  const rawCalls = () => fetchFn.mock.calls.filter(([u]) => String(u) === '/api/events/9/raw').length

  await user.click(within(item).getByRole('button', { name: CARD_NOTICE.showRaw })) // 1) 펼침 → 로딩
  expect(rawCalls()).toBe(1)
  await user.click(within(item).getByRole('button', { name: CARD_NOTICE.hideRaw })) // 2) 로딩 중 접기
  reject(new TypeError('Failed to fetch')) // 3) 첫 요청 실패 (접힌 채 error 상태)
  await waitFor(() => expect(within(item).getByRole('button', { name: CARD_NOTICE.showRaw })).toBeInTheDocument())

  console.log('[P3] 접힌 채 실패 뒤 — alert 보임 =', within(item).queryByRole('alert') !== null)

  await user.click(within(item).getByRole('button', { name: CARD_NOTICE.showRaw })) // 4) 다시 펼치기

  expect(within(item).queryByTestId('raw-utterance')).not.toBeNull()
