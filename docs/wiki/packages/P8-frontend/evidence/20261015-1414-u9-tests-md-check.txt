const verbose = readFileSync(`${ROOT}/evidence/20261015-1410-u9-vitest-verbose.txt`, 'utf-8').split('\n')
const md = readFileSync(`${ROOT}/tests.md`, 'utf-8').split('\n')

const actual = []
for (const line of verbose) {

  if (m) actual.push(`${m[1]}::${m[2].split(' > ').pop()}`)

const rows = []
const rowIds = {}
for (const line of md) {
  const m = /^\| ((?:\w+)\.test\.tsx?)::(.+?) \| /.exec(line)
  if (m) {
    const key = `${m[1]}::${m[2]}`
    rows.push(key)

    const result = cols[cols.length - 1]
    rowIds[key] = [...result.matchAll(/U\d-[MVRXP]\d+/g)].map((x) => x[0])
  }

  const hits = lines.filter((l) => l.startsWith(` ✓ src/__tests__/${file} > `) && l.replace(/ \d+ms$/, '').endsWith(` > ${name}`))

// 기존 변이 증거에서 "어느 변이가 어느 테스트를 실패시켰나" 를 뽑는다 (읽기 전용). 출력: JSON { 테스트이름: ["U8-M3", ...] }
import { readFileSync, readdirSync } from 'node:fs'
import { join } from 'node:path'

const DIR = '/Users/sunwoo/Desktop/Portfolio/Relationship/docs/wiki/packages/P8-frontend/evidence'
const files = readdirSync(DIR).filter((f) => /(u[4-9])-.*mutation/.test(f) && !/script|raw/.test(f))
const map = {}
const survived = {}
const detail = []

for (const f of files.sort()) {
  const unit = /-(?:verifier-)?(u\d)/.exec(f)[1].toUpperCase()
  const verifier = f.includes('verifier')
  const lines = readFileSync(join(DIR, f), 'utf-8').split('\n')
  let id = null
  for (const line of lines) {
    let m
    if ((m = /^#{3,5} (?:변이:? ?)?([A-Z]\d+)\b/.exec(line))) id = m[1]
    else if ((m = /^={4,} ([A-Z]\d+)\b/.exec(line))) id = m[1]
    else if ((m = /^# 명령: .*[ '"]([A-Z]\d+)-/.exec(line))) id = m[1]
    else if ((m = /^# 명령: .*mut_u\d\.(?:py|mjs) ([A-Z]\d+)\s*$/.exec(line))) id = m[1]
    const x = /^\s+× (.+?) \d+ms$/.exec(line)
    if (x && id && !/^[A-Z]0$/.test(id) && !/^E\d/.test(id)) {
      const tag = `${unit}-${id}`
      ;(map[x[1]] ??= new Set()).add(tag)
      detail.push([f, tag, x[1]])
    }
  }
}
const out = Object.fromEntries(Object.entries(map).map(([k, v]) => [k, [...v].sort()]))
console.log(JSON.stringify({ files, out }, null, 1))
