ROUTE_NAMES = ("get_event_raw", "get_fact_sources", "get_person_card", "list_briefings", "list_persons")


def _apply_read_mut(name: str) -> None:
    import app.api.read as read
    import app.api.routes as routes

    desc, old, new = READ_MUTS[name]
    src = Path(read.__file__).read_text(encoding="utf-8")
    n = src.count(old)
    expected = 2 if name == "U1R" else 1  # U1R: 목록·카드 두 곳의 max 를 모두 min 으로
    if n != expected:
        raise SystemExit(f"[mutplug] {name}: 원문 등장 횟수 {n} ({expected} 이어야 함)")
    src = src.replace(old, new)
    if name == "V6":
        src = src.replace("from sqlalchemy import BigInteger, func, select", "from sqlalchemy import BigInteger, func, select, true")
    code = compile(src, read.__file__ + f"#{name}", "exec")
    exec(code, read.__dict__)
    for attr in ROUTE_NAMES:
        setattr(routes, attr, getattr(read, attr))
    print(f"[mutplug] {name} 적용: {desc}")


def _apply_pending_model_mut() -> None:
    """V8 [표에 없음]: 대기 질문 응답 모델에서 `kind` 필드를 뺀다 (pydantic 이 모르는 키를 버림)."""
    import app.api.routes as routes
    from pydantic import BaseModel

    class ChatPendingQuestionOutNoKind(BaseModel):
        question_id: int
        status: str
        question: str
        options: list[str]

    routes.ChatPendingQuestionOut = ChatPendingQuestionOutNoKind  # type: ignore[attr-defined]
    print("[mutplug] V8 적용: ChatPendingQuestionOut 에서 kind 필드 누락")


def pytest_configure(config):  # noqa: D103
    if not MUT:
        print("[mutplug] 변이 없음(대조군)")
        return
    if MUT in READ_MUTS:
        _apply_read_mut(MUT)
    elif MUT == "V8":
        _apply_pending_model_mut()
    else:
        raise SystemExit(f"[mutplug] 모르는 변이 {MUT}")
