import { cpSync, existsSync, readFileSync, rmSync, symlinkSync, writeFileSync } from 'node:fs'
import { spawnSync } from 'node:child_process'
import { join } from 'node:path'

const SRC = '/Users/sunwoo/Desktop/Portfolio/Relationship/web'

function fresh() {
  if (existsSync(DST)) rmSync(DST, { recursive: true, force: true })
  cpSync(SRC, DST, { recursive: true, filter: (p) => !/(node_modules|[\\/]dist|test-results)([\\/]|$)/.test(p.slice(SRC.length)) })
  symlinkSync(join(SRC, 'node_modules'), join(DST, 'node_modules'))
}

function sub(path, oldText, newText) {
  const p = join(DST, path)
  const s = readFileSync(p, 'utf-8')
  if (!s.includes(oldText)) throw new Error(`치환 대상 없음: ${JSON.stringify(oldText)}`)
  writeFileSync(p, s.replace(oldText, newText))
}
