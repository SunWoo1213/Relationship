def run(name, rel, old, new):
    d = os.path.join(BASE, "work")
    if os.path.exists(d):
        shutil.rmtree(d)
    os.makedirs(d)
    for item in ["app", "tests", "alembic", "alembic.ini", "pyproject.toml"]:
        src = os.path.join(REPO, item)
        dst = os.path.join(d, item)
        if os.path.isdir(src):
            shutil.copytree(src, dst, ignore=shutil.ignore_patterns("__pycache__", ".pytest_cache"))
        else:
            shutil.copy(src, dst)
    if rel:
        p = os.path.join(d, rel)
        text = open(p, encoding="utf-8").read()
        if old not in text:
            return f"{name}: !! 원문 못 찾음"
        open(p, "w", encoding="utf-8").write(text.replace(old, new))
    env = dict(os.environ, POSTGRES_PORT="5433")
    out = subprocess.run(
        [PY, "-m", "pytest", "-q", "-p", "no:cacheprovider", *TESTS],
        cwd=d, env=env, capture_output=True, text=True,
    ).stdout
    lines = [l for l in out.splitlines() if l.startswith("FAILED") or " passed" in l or " failed" in l or "error" in l.lower()]
    return f"{name}\n" + "\n".join("    " + l for l in lines)
